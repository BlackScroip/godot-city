cmake_minimum_required(VERSION 3.22)
project(city_combat_ai LANGUAGES CXX)

if(NOT DEFINED GODOT_CPP_SOURCE OR NOT DEFINED CITY_EXTENSION_API)
  message(FATAL_ERROR "Set GODOT_CPP_SOURCE and CITY_EXTENSION_API to temporary dependency/API paths")
endif()
set(GODOTCPP_CUSTOM_API_FILE "${CITY_EXTENSION_API}" CACHE FILEPATH "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
set(CITY_TARGET template_release CACHE STRING "Godot template_debug or template_release")
set(GODOTCPP_TARGET "${CITY_TARGET}" CACHE STRING "" FORCE)
set(GODOTCPP_PRECISION single CACHE STRING "" FORCE)
set(GODOTCPP_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${GODOT_CPP_SOURCE}" godot-cpp)
add_library(combat_ai SHARED src/combat_ai_probe.cpp src/register_types.cpp)
target_link_libraries(combat_ai PRIVATE godot-cpp)
target_compile_features(combat_ai PRIVATE cxx_std_17)
target_compile_options(combat_ai PRIVATE -O3 -fno-fast-math -ffp-contract=off)
set_target_properties(combat_ai PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
